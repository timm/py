1 #!/usr/bin/env python3   2 """   3 egnb: naive bayes. Learn one Tab per class; classify a row by   4 which class's columns like it most. Train on each row after   5 predicting it, so accuracy is on unseen rows. Data: --file   6 names in $MOOT/classify (with a class! column).   7 (c) 2026, Tim Menzies <timm@ieee.org>, MIT license   8 """   9 import os, sys  # make hair importable whether run from source or installed  10 sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))  11 from math import log, pi, exp  12 from hair import *  13   14 DATA   = "diabetes soybean breast.w vote".split()  15 SMOOTH = [1, 2, 3]                                    # m and k each range here  16 EVALS  = {"auc": (aucs, True),                        # bigger = better  17           "acc": (lambda ps: 100*sum(w==max(p,key=p.get) for w,p in ps)  18                                /(len(ps)+TINY), True)}  19   20 def like(col, v, prior: float, m: float) -> float:  21   "log P(v|class); gaussian for Nums, m-estimate (weight m) for Syms"  22   if isa(col, Sym):  23     N = sum(col.values())  24     return log((col.get(v,0) + m*prior) / (N + m + TINY))  25   n, mu, _ = col  26   sd = div(col) + TINY  27   return -(v - mu)**2/(2*sd*sd) - log(sd * (2*pi)**.5)  28   29 def logp(tab, row, n, nh, m, k) -> float:  30   "log posterior score of one class. k = prior Laplace, m = feature m-estimate."  31   prior = (len(tab.rows) + k) / (n + k*nh)  32   return log(prior) + sum(like(tab.cols[a], row[a], prior, m)  33                           for a in tab.x if row[a] != "?")  34   35 def posts(tabs: dict, row: list, n, m, k) -> dict:  36   "Normalized posteriors {class: prob}; subtract max-log so exp won't underflow."  37   ls = {c: logp(t, row, n, len(tabs), m, k) for c, t in tabs.items()}  38   hi = max(ls.values())  39   es = {c: exp(v - hi) for c, v in ls.items()}  40   z  = sum(es.values()) + TINY  41   return {c: e/z for c, e in es.items()}  42   43 def nb(tbl, train, test):  44   "Yield (smoothing, want, posteriors) per test row, m-estimate x Laplace."  45   tabs = {}  46   for row in train:                               # train: one Tab per class  47     c = row[tbl.klass]; tabs.setdefault(c, clone(tbl)); addRow(tabs[c], row)  48   for row in test:  49     for m in SMOOTH:  50       for k in SMOOTH:  51         yield f"m={m}:k={k}", row[tbl.klass], posts(tabs, row, len(train), m, k)  52   53 def main() -> None:  54   cli(the)  55   metric   = the.eval if the.eval in EVALS else "auc"  56   err, big = EVALS[metric]  57   fs = ["$MOOT/classify/%s.csv" % f for f in DATA]  58   print(conf(the, "nb.py", eval=metric, xval="5x5"))  59   print("# %s (weighted, 0..100), bigger=better; ! marks rank-1; last col = data set\n"  60         % metric)  61   learn(fs, nb, err, reverse=big)  62   63 if __name__ == "__main__": main()  64
