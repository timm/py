1 #!/usr/bin/env python3   2 """   3 knn: guess a row's disty from its k nearest rows.   4 Table 1: kernels flat (all neighbours count the same) and tri   5 (nearer counts more: weight 1 - d/dmax) at k=1..5, plus mu   6 (always guess the train mean). Table 2: flat kernel at k=4,   7 trained on 32..512 rows, or all, on BIGS --dir csvs with ROWS   8 rows. Both: 5 repeats of a 2-way cross-val, scored by RAE, on at   9 most --test test rows and --pool train rows.  10 (c) 2026, Tim Menzies <timm@ieee.org>, MIT license  11 """  12 import os, sys  # make hair importable from source or installed  13 sys.path.insert(0, os.path.dirname(os.path.dirname(  14                    os.path.abspath(__file__))))  15 import random  16 from glob import glob  17 from hair import *  18   19 BEST_K = 4                        # table 1's pick, used by table 2  20 SIZES  = [32, 64, 128, 256, 512]  # train sizes for table 2  21 ROWS   = range(1000, 10000)       # table 2: sets with these #rows  22 BIGS   = 20                       # table 2: how many such sets  23   24 #-- kernels: guess row's disty from rows (nearest first) ------------  25 def flat(tbl, rows: list, row=None) -> float:  26   return mid(adds(disty(tbl, r) for r in rows))  27   28 def tri(tbl, rows: list, row: list) -> float:  29   ds = [distx(tbl, r, row) for r in rows]  30   ws = [1 - d/(max(ds) + TINY) for d in ds]  31   return (sum(w*disty(tbl, r) for w, r in zip(ws, rows))  32           / (sum(ws) + TINY))  33   34 #-- learner: each treatment t = o(name, f, k, n) --------------------  35 def T(name, f=None, k=None, n=None) -> o:  36   "Kernel f (default flat) on the k nearest of the first n train rows."  37   return o(name=name, f=f or flat, k=k, n=n)   # None = all  38   39 def big() -> list:  40   "BIGS random (by --seed) csvs under --dir with ROWS rows."  41   d  = os.path.expanduser(the.dir.replace("$MOOT", MOOT, 1))  42   fs = sorted(f for f in glob(f"{d}/**/*.csv", recursive=True)  43               if sum(1 for _ in open(f)) - 1 in ROWS)  44   return random.Random(the.seed).sample(fs, min(BIGS, len(fs)))  45   46 def knn(ts: list):  47   "Learner: sort train once per test row; every t slices that sort."  48   def learner(tbl, train, test):  49     train, test = train[:the.pool], test[:the.test]  # caps: speed  50     pos = {id(r): i for i, r in enumerate(train)}  # xval shuffled train  51     for row in test:  52       want, ns = disty(tbl, row), near(tbl, train, row, len(train))  53       for t in ts:  54         rows = [r for r in ns if t.n is None or pos[id(r)] < t.n]  55         yield t.name, want, t.f(tbl, rows[:t.k], row)  56   return learner  57   58 def main() -> None:  59   cli(the)                         # so the config below reflects flags  60   print(conf(the, "knn.py", eval="rae", xval="5x2"))  61   print("# rae: 100 = no better than guessing the mean; lower=better\n"  62         "# ! marks a rank-1 (best) cell; last col = data set")  63   for x in [o(title = "kernels x k",   64               files = files(),  65               ts    = [T(f"{f.__name__}:k={k}", f, k)  66                        for f in (flat, tri) for k in range(1, the.k+1)]  67                       + [T("mu")]  68               ),           # mu: flat over all train  69             o(title = f"flat:k={BEST_K} x train size (all = pool)",  70               files = big(),  71               ts    = [T(f"n={n or 'all'}", k=BEST_K, n=n)  72                        for n in SIZES + [None]]  73               )]:  74     print(f"\n# {x.title}\n")  75     learn(x.files, knn(x.ts), rae, n=2)  76   77 if __name__ == "__main__": main()  78
